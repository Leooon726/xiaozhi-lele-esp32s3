cmake_minimum_required(VERSION 3.20)
project(lele_touch_hit_test LANGUAGES C CXX ASM)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LELE_ROOT}/managed_components/lvgl__lvgl" lvgl)
add_executable(touch_hit_test touch_hit_test.cc)
target_compile_features(touch_hit_test PRIVATE cxx_std_17)
target_compile_options(touch_hit_test PRIVATE -UNDEBUG)
target_include_directories(touch_hit_test PRIVATE "${LELE_ROOT}/main/boards/lckfb/szpi-esp32s3")
target_link_libraries(touch_hit_test PRIVATE lvgl::lvgl)
